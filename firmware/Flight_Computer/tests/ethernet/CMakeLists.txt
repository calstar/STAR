cmake_minimum_required(VERSION 3.16)
project(star_ethernet LANGUAGES C CXX)
enable_testing()
option(SELF_TEST_SANITIZERS "Address/undefined sanitizers" OFF)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(PROTOCOL ${ROOT}/../../lib/daq-protocol/src)
add_library(ethernet STATIC ${ROOT}/chips/w5500_udp.c ${ROOT}/app/daq_network.c ${ROOT}/app/daq_wire.c ${ROOT}/app/daq_records.c)
target_include_directories(ethernet PUBLIC ${ROOT}/chips ${ROOT}/app)
add_library(daq_reference STATIC ${PROTOCOL}/DiabloPacketUtils.cpp)
target_include_directories(daq_reference PUBLIC ${PROTOCOL})
add_executable(test_w5500 test_w5500.c)
add_executable(test_task test_task.c ${ROOT}/app/daq_task.c)
target_include_directories(test_task BEFORE PRIVATE stubs)
target_link_libraries(test_task PRIVATE ethernet)
add_executable(test_records test_records.c)
add_executable(test_wire test_wire.cpp)
target_link_libraries(test_w5500 PRIVATE ethernet)
target_link_libraries(test_records PRIVATE ethernet)
target_link_libraries(test_wire PRIVATE ethernet daq_reference)
add_executable(test_network test_network.c)
target_link_libraries(test_network PRIVATE ethernet)
add_executable(test_defaults test_task.c ${ROOT}/app/daq_task.c)
target_include_directories(test_defaults BEFORE PRIVATE stubs)
target_compile_definitions(test_defaults PRIVATE TEST_DEFAULTS)
target_link_libraries(test_defaults PRIVATE ethernet)
foreach(t ethernet test_network test_defaults test_task test_w5500 test_records test_wire)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
foreach(t ethernet daq_reference test_network test_defaults test_task test_w5500 test_records test_wire)
  set_target_properties(${t} PROPERTIES C_STANDARD 11 CXX_STANDARD 17)
  if(SELF_TEST_SANITIZERS)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME w5500 COMMAND test_w5500)
add_test(NAME records COMMAND test_records)
add_test(NAME wire_compatibility COMMAND test_wire)

add_test(NAME network_task COMMAND test_task)

add_test(NAME discovery_identity COMMAND test_network)
add_test(NAME disabled_profile COMMAND test_defaults)
