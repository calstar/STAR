cmake_minimum_required(VERSION 3.16)
project(star_fusion LANGUAGES C)
enable_testing()
option(SELF_TEST_SANITIZERS "Address and undefined behavior sanitizers" OFF)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_library(fusion STATIC ${ROOT}/src/app/fusion.c)
target_include_directories(fusion PUBLIC ${ROOT}/src/app)
add_executable(test_fusion test_fusion.c)
add_executable(fusion_replay replay.c)
target_link_libraries(test_fusion PRIVATE fusion)
target_link_libraries(fusion_replay PRIVATE fusion)
if(UNIX)
    target_link_libraries(fusion PUBLIC m)
endif()
add_test(NAME fusion_units COMMAND test_fusion)
foreach(t fusion test_fusion fusion_replay)
    set_target_properties(${t} PROPERTIES C_STANDARD 11 C_EXTENSIONS OFF)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(SELF_TEST_SANITIZERS)
        target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
