cmake_minimum_required(VERSION 3.16)
project(star_self_test LANGUAGES C)

enable_testing()
option(SELF_TEST_SANITIZERS "Run host tests with address/undefined sanitizers" OFF)

add_library(star_self_test STATIC ../../src/app/self_test.c ../../src/app/self_test_devices.c)
target_include_directories(star_self_test PUBLIC ../../src/app ../../src/chips)
target_compile_features(star_self_test PUBLIC c_std_99)
set_target_properties(star_self_test PROPERTIES C_EXTENSIONS OFF)
add_executable(test_self_test test_self_test.c)
target_link_libraries(test_self_test PRIVATE star_self_test)

add_executable(test_self_test_devices test_self_test_devices.c)
target_link_libraries(test_self_test_devices PRIVATE star_self_test)

# Portable protocol helpers used by the board layer (no HAL).
add_library(star_board_proto STATIC ../../src/chips/gnss_frame.c ../../src/chips/sd_spi.c)
target_include_directories(star_board_proto PUBLIC ../../src/app ../../src/chips)
target_compile_features(star_board_proto PUBLIC c_std_99)
set_target_properties(star_board_proto PROPERTIES C_EXTENSIONS OFF)
add_executable(test_gnss_frame test_gnss_frame.c)
target_link_libraries(test_gnss_frame PRIVATE star_board_proto)
add_executable(test_sd_spi test_sd_spi.c)
target_link_libraries(test_sd_spi PRIVATE star_board_proto)

foreach(target star_self_test test_self_test test_self_test_devices star_board_proto test_gnss_frame test_sd_spi)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(SELF_TEST_SANITIZERS)
        if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "Sanitizers require Clang or GCC")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
add_test(NAME self_test_contract COMMAND test_self_test)

add_test(NAME self_test_device_protocols COMMAND test_self_test_devices)
add_test(NAME gnss_frame COMMAND test_gnss_frame)
add_test(NAME sd_spi COMMAND test_sd_spi)
